public frm_snackInvent()
{
    InitializeComponent(); // Initializes the form components
    toolStripStatusLabel1.Text = ""; // Wipe the status strip of all text
}

private void bttn_snackImage_Click(object sender, EventArgs e)
{
    // Set the filter for the open file dialog to allow only image files
    openFileDialog1.Filter = "Image Files(*.jpg;*.jpeg;*.png;*.gif;*bmp)|*.jpg;*.jpeg;*.png;*.gif;*.bmp";
    
    // Show the open file dialog and check if the user selected a file
    if(openFileDialog1.ShowDialog() == DialogResult.OK) { 
        txt_imagePath.Text = openFileDialog1.FileName; // Set the text box to the selected file path
        pb_snackIcon.ImageLocation = openFileDialog1.FileName; // Display the selected image in the picture box
    }
}

private void btn_add_Click(object sender, EventArgs e)
{
    // Check if the image path is valid
    if(!File.Exists(txt_imagePath.Text))
    {
        MessageBox.Show("Please Select an image"); // Prompt user to select an image
        bttn_snackImage.Focus(); // Set focus back to the image button
    } 
    // Check if the snack name is provided
    else if(txt_snackName.Text.Length == 0)
    {
        MessageBox.Show("Please Input a name"); // Prompt user to input a name
        txt_snackName.Focus(); // Set focus back to the snack name text box
    } 
    // Check if the snack price is greater than zero
    else if(nud_snackPrice.Value == 0)
    {
        MessageBox.Show("Please enter a price that is more than 0"); // Prompt user for a valid price
        nud_snackPrice.Focus(); // Set focus back to the price numeric up-down
    } 
    else
    {
        // Create a new snack object and populate its properties
        snackInvent snack = new snackInvent();
        snack.name = txt_snackName.Text; // Assign the snack name
        snack.price = Convert.ToDouble(nud_snackPrice.Value); // Assign the snack price
        snack.amount = Convert.ToInt32(nud_snackQuantity.Value); // Assign the snack quantity
        snack.imagepath = txt_imagePath.Text; // Assign the image path
        
        // Add the new snack to the global snacks list
        Program.snacks.Add(snack);

        // Add the snack details to the data grid view
        dgv_invent.Rows.Add(snack.name, snack.price, snack.amount, Image.FromFile(snack.imagepath), snack.imagepath);

        // Reset the input fields for the next entry
        txt_imagePath.Text = ""; // Clear the image path text box
        txt_snackName.Text = ""; // Clear the snack name text box
        nud_snackPrice.Value = 0; // Reset the price numeric up-down
        nud_snackQuantity.Value = 0; // Reset the quantity numeric up-down
        pb_snackIcon.Image = null; // Clear the displayed image
    }
}

private void frm_snackInvent_FormClosing(object sender, FormClosingEventArgs e)
{
    try
    {
        // Create a StreamWriter to write the snacks to a CSV file
        using (StreamWriter sw = new StreamWriter("snacks.csv")) { 
            foreach(var snack in Program.snacks)
            {
                // Write each snack's details to the CSV file
                sw.WriteLine(snack.name + "," + snack.price + "," + snack.amount + "," + snack.imagepath);
            }
        }
    }
    catch (Exception ex)
    {
        // Show an error message if there is an issue during file write
        MessageBox.Show("Error during file write", "", MessageBoxButtons.OK, MessageBoxIcon.Error);
    }
}

private void frm_snackInvent_Load(object sender, EventArgs e)
{
    try
    {
        // Check if the snacks CSV file exists
        if(File.Exists("snacks.csv"))
        {
            // Create a StreamReader to read the snacks from the CSV file
            using (StreamReader sr = new StreamReader("snacks.csv"))
            {
                // Read each line until the end of the file
                while(!sr.EndOfStream)
                {
                    snackInvent snack = new snackInvent(); // Create a new snack object
                    // Read a line from the CSV file
                    string line = sr.ReadLine();
                    // Split the line into an array based on commas
                    string[] arr = line.Split(',');
                    snack.name = arr[0]; // Assign the snack name
                    snack.price = double.Parse(arr[1]); // Assign the snack price
                    snack.amount = Int32.Parse(arr[2]); // Assign the snack quantity
                    snack.imagepath = arr[3]; // Assign the image path

                    // Add the snack to the global snacks list
                    Program.snacks.Add(snack);
                    // Add the snack details to the data grid view
                    dgv_invent.Rows.Add(snack.name, snack.price, snack.amount, Image.FromFile(snack.imagepath), snack.imagepath);
                }
            }
        }
    } 
    catch (Exception ex)
    {
        // Show an error message if there is an issue during file read
        MessageBox.Show("Error during file read", "", MessageBoxButtons.OK, MessageBoxIcon.Error);
    }
}
